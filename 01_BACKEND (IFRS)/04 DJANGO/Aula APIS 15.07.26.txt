import contextlib, io  # importanto a bibliotecas
from manage import *

# define que a saída será o terminal
saida = io.StringIO  # a saída é do tipo string


with contextlib.redirect_stdout(saida): # pega a biblioteca de contexto e redireciona a saída padrão para saída
	main() #com essa definição executa o main
